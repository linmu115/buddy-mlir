cmake_minimum_required(VERSION 3.20)
project(RouteBHost LANGUAGES CXX)
set(pipeline "${CMAKE_CURRENT_LIST_DIR}/../..")
get_filename_component(repo "${pipeline}/../../.." ABSOLUTE)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(RISCV_MPI_DIR)
  add_library(route_b_mpi INTERFACE)
  target_include_directories(route_b_mpi INTERFACE "${RISCV_MPI_DIR}/include")
  target_link_libraries(route_b_mpi INTERFACE "${RISCV_MPI_DIR}/lib/libmpi.so")
else()
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_library(route_b_mpi ALIAS MPI::MPI_CXX)
endif()
add_library(route_b_stage STATIC "${pipeline}/TemplateStage.cpp" "${pipeline}/TemplateBundle.cpp")
target_compile_features(route_b_stage PUBLIC cxx_std_17)
target_include_directories(route_b_stage PUBLIC "${repo}/runtime/include" "${pipeline}")
target_link_libraries(route_b_stage PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_executable(buddy-pp-route-b "${pipeline}/RouteBMain.cpp")
target_include_directories(buddy-pp-route-b PRIVATE "${repo}/frontend/Interfaces")
target_link_libraries(buddy-pp-route-b PRIVATE route_b_stage route_b_mpi)
include(CTest)
if(BUILD_TESTING AND NOT RISCV_MPI_DIR)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(tests "${repo}/tests/Pipeline")
  add_library(route_b_mpi_trace SHARED "${tests}/RouteBMPITrace.cpp")
  target_link_libraries(route_b_mpi_trace PRIVATE route_b_mpi)
  target_include_directories(route_b_mpi_trace PRIVATE "${repo}/runtime/include")
  foreach(role RANGE 1 3)
    add_library(route_b_mock_${role} SHARED "${tests}/RouteBMock.cpp")
    target_compile_definitions(route_b_mock_${role} PRIVATE MOCK_ROLE=${role})
    target_include_directories(route_b_mock_${role} PRIVATE "${repo}/runtime/include")
  endforeach()
  add_library(route_b_mock_incomplete SHARED "${tests}/RouteBMock.cpp")
  target_compile_definitions(route_b_mock_incomplete PRIVATE MOCK_ROLE=2 MOCK_INCOMPLETE)
  target_include_directories(route_b_mock_incomplete PRIVATE "${repo}/runtime/include")
  add_executable(route_b_local "${tests}/RouteBLocal.cpp")
  target_link_libraries(route_b_local PRIVATE route_b_stage)
  foreach(test storage_and_dso local_stage mpi_ranges mpi_wire)
    add_test(NAME test_route_b_${test} COMMAND "${Python3_EXECUTABLE}"
      "${tests}/test_route_b_host.py" "${test}" "${CMAKE_CURRENT_BINARY_DIR}" "${MPIEXEC_EXECUTABLE}")
    set_tests_properties(test_route_b_${test} PROPERTIES TIMEOUT 180)
  endforeach()
endif()
